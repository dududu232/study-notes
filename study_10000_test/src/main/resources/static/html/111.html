html转pdf（总结五种方法Java）

Java 实现html转pdf，总结五种方法。
推荐使用wkhtmltopdf,Itext

 

方法一：使用wkhtmltopdf

1、下载插件wkhtmltopdf
https://wkhtmltopdf.org/downloads.html

2、本机测试
本目录下cmd进入
输入命令 wkhtmltopdf.exe ‪E:\学习文档\百度常用标签.html ‪E:\学习文档\百度常用标签.pdf

  7.  
  8.     /**
  9.      * html转pdf
  10.      *
  11.      * @param srcPath  html路径，可以是硬盘上的路径，也可以是网络路径
  12.      * @param destPath pdf保存路径
  13.      * @return 转换成功返回true
  14.      */
  15.     public static boolean convert(String srcPath, String destPath,String toPdfTool){
  16.         File file = new File(destPath);
  17.         File parent = file.getParentFile();
  18.         //如果pdf保存路径不存在，则创建路径
  19.         if(!parent.exists()){
  20.             parent.mkdirs();
  21.         }
  22.         StringBuilder cmd = new StringBuilder();
  23.         cmd.append(toPdfTool);
  24.         cmd.append(" ");
  25.         cmd.append(" --header-line");//页眉下面的线
  26.         cmd.append(" --margin-top 3cm ");//设置页面上边距 (default 10mm)
  27.         // cmd.append(" --header-html file:///"+WebUtil.getServletContext().getRealPath("")+FileUtil.convertSystemFilePath("\\style\\pdf\\head.html"));// (添加一个HTML页眉,后面是网址)
  28.         cmd.append(" --header-spacing 5 ");// (设置页眉和内容的距离,默认0)
  29.         //cmd.append(" --footer-center (设置在中心位置的页脚内容)");//设置在中心位置的页脚内容
  30.         //cmd.append(" --footer-html file:///"+WebUtil.getServletContext().getRealPath("")+FileUtil.convertSystemFilePath("\\style\\pdf\\foter.html"));// (添加一个HTML页脚,后面是网址)
  31.         cmd.append(" --footer-line");//* 显示一条线在页脚内容上)
  32.         cmd.append(" --footer-spacing 5 ");// (设置页脚和内容的距离)
  33.         cmd.append(srcPath);
  34.         cmd.append(" ");
  35.         cmd.append(destPath);
  36.         boolean result = true;
  37.         try{
  38.             Process proc = Runtime.getRuntime().exec(cmd.toString());
  39.             HtmlToPdfInterceptor error = new HtmlToPdfInterceptor(proc.getErrorStream());
  40.             HtmlToPdfInterceptor output = new HtmlToPdfInterceptor(proc.getInputStream());
  41.             error.start();
  42.             output.start();
  43.             proc.waitFor();
  44.         }catch(Exception e){
  45.             result = false;
  46.             e.printStackTrace();
  47.         }
  48.  
  49.         return result;
  50.     }

  1. /**
  2.  * Itext7转换工具类
  3.  */
  4. @Slf4j
  5. public class HtmlToPdfUtils {
  6.  
  7.     /**
  8.      * html转pdf
  9.      *
  10.      * @param inputStream  输入流
  11.      * @param waterMark    水印
  12.      * @param fontPath     字体路径，ttc后缀的字体需要添加<b>,0<b/>
  13.      * @param outputStream 输出流
  14.      * @date : 2022/11/15 14:07
  15.      */
  16.     public static void convertToPdf(InputStream inputStream, String waterMark, String fontPath, OutputStream outputStream) throws IOException {
  17.  
  18.         PdfWriter pdfWriter = new PdfWriter(outputStream);
  19.         PdfDocument pdfDocument = new PdfDocument(pdfWriter);
  20.         //设置为A4大小
  21.         pdfDocument.setDefaultPageSize(PageSize.A4);
  22.         //添加水印
  23.         pdfDocument.addEventHandler(PdfDocumentEvent.END_PAGE, new WaterMarkEventHandler(waterMark));
  24.  
  25.         //添加中文字体支持
  26.         ConverterProperties properties = new ConverterProperties();
  27.         FontProvider fontProvider = new FontProvider();
  28.  
  29.         //        设置字体
  30.         /*PdfFont sysFont = PdfFontFactory.createFont("STSongStd-Light", "UniGB-UCS2-H", false);
  31.         fontProvider.addFont(sysFont.getFontProgram(), "UniGB-UCS2-H");*/
  32.  
  33.         //添加自定义字体，例如微软雅黑
  34.         if (StringUtils.isNotBlank(fontPath)) {
  35.             PdfFont microsoft = PdfFontFactory.createFont(fontPath, PdfEncodings.IDENTITY_H, false);
  36.             fontProvider.addFont(microsoft.getFontProgram(), PdfEncodings.IDENTITY_H);
  37.         }
  38.  
  39.         properties.setFontProvider(fontProvider);
  40.         //        读取Html文件流，查找出当中的&nbsp;或出现类似的符号空格字符
  41.         inputStream = readInputStrem(inputStream);
  42.         if (inputStream != null) {
  43.             //        生成pdf文档
  44.             HtmlConverter.convertToPdf(inputStream, pdfDocument, properties);
  45.             pdfWriter.close();
  46.             pdfDocument.close();
  47.             return;
  48.         } else {
  49.             log.error("转换失败！");
  50.         }
  51.     }
  52.  
  53.     /**
  54.      * 读取HTML 流文件，并查询当中的&nbsp;或类似符号直接替换为空格
  55.      *
  56.      * @param inputStream
  57.      * @return
  58.      */
  59.     private static InputStream readInputStrem(InputStream inputStream) {
  60.         // 定义一些特殊字符的正则表达式 如：
  61.         String regEx_special = "\\&[a-zA-Z]{1,10};";
  62.         try {
  63.             //<1>创建字节数组输出流，用来输出读取到的内容
  64.             ByteArrayOutputStream baos = new ByteArrayOutputStream();
  65.             //<2>创建缓存大小
  66.             byte[] buffer = new byte[1024]; // 1KB
  67.             //每次读取到内容的长度
  68.             int len = -1;
  69.             //<3>开始读取输入流中的内容
  70.             while ((len = inputStream.read(buffer)) != -1) { //当等于-1说明没有数据可以读取了
  71.                 baos.write(buffer, 0, len);   //把读取到的内容写到输出流中
  72.             }
  73.             //<4> 把字节数组转换为字符串
  74.             String content = baos.toString();
  75.             //<5>关闭输入流和输出流
  76.             //            inputStream.close();
  77.             baos.close();
  78.             //            log.info("读取的内容：{}", content);
  79.             //            判断HTML内容是否具有HTML的特殊字符标记
  80.             Pattern compile = Pattern.compile(regEx_special, Pattern.CASE_INSENSITIVE);
  81.             Matcher matcher = compile.matcher(content);
  82.             String replaceAll = matcher.replaceAll("");
  83.             //            log.info("替换后的内容：{}", replaceAll);
  84.             //            将字符串转化为输入流返回
  85.             InputStream stringStream = getStringStream(replaceAll);
  86.             //<6>返回结果
  87.             return stringStream;
  88.         } catch (Exception e) {
  89.             e.printStackTrace();
  90.             log.error("错误信息：{}", e.getMessage());
  91.             return null;
  92.         }
  93.     }
  94.  
  95.     /**
  96.      * 将一个字符串转化为输入流
  97.      * @param sInputString 字符串
  98.      * @return
  99.      */
  100.     public static InputStream getStringStream(String sInputString) {
  101.         if (sInputString != null && !sInputString.trim().equals("")) {
  102.             try {
  103.                 ByteArrayInputStream tInputStringStream = new ByteArrayInputStream(sInputString.getBytes());
  104.                 return tInputStringStream;
  105.             } catch (Exception e) {
  106.                 e.printStackTrace();
  107.             }
  108.         }
  109.         return null;
  110.     }
  111.  
  112. }

方法三：使用Spire.Doc

  1. 将文档从一种格式转换为另一种格式是Spire.Doc的主要功能之一。这种转换只不过是加载和保存操作的组合。因此，使用Spire.DOC可以将文档从任何受支持的加载格式转换为任何受支持的保存格式。
  2. spire.doc分为商业版和免费版，免费版只支持转换前3页，以免费版为例
